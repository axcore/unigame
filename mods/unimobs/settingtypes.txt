# Misc item settings
# Non-randomised interval after which thrown cobwebs disappear, in seconds
unimobs_petz_cobweb_decay_interval (Cobweb decay interval in seconds) int 1200
# Used with tamed mobs which have dreamcatchers, when the player has enabled the "Automatic go home"
# button in the main mob formspec. If the distance between the specified home position and the mob's
# current position is greater than this value, then the mob is automatically teleported back to the
# specified home position
unimobs_petz_teleport_back_home_distance (Dreamcatcher teleport distance) int 50
# If true, placement of mob shelters is disabled (but its constituent blocks will still exist in the
# game). In the original code, this setting was used only with dog kennels
unimobs_petz_disable_shelter_placement_flag (Disable shelter placement) bool false

# Curse of Lycanthropy settings

# Chance of a wolf infecting a player with the Curse of Lycanthropy
unimobs_petz_lycanthropy_wolf_infection_chance (Chance of wolves infecting players) float 200
# Chance of a werewolf infecting a player with the Curse of Lycanthropy
unimobs_petz_lycanthropy_werewolf_infection_chance (Chance of werewolves infecting players) float 10
# Players infected with the Curse of Lycanthropy take less damage; a value of 0.5 represents a
# reduction of 50% in the amount of damage taken
unimobs_petz_lycanthropy_damage_reduction_factor (Damage reduction factor for infected players) float 0.5
